CREATE DATABASE Autószerelő
DEFAULT charset utf8
COLLATE utf8_hungarian_ci

CREATE TABLE `ugyfel` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    `nev` VARCHAR(255) NOT NULL,
    `telefonszam` VARCHAR(20) NOT NULL,
    `lakcim` VARCHAR(255) NOT NULL
);

CREATE TABLE `auto` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    `rendszam` VARCHAR(255) NOT NULL UNIQUE,
    `ugyfelId` BIGINT UNSIGNED NOT NULL
);

CREATE TABLE `alkatresz` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    `nev` VARCHAR(255) NOT NULL,
    `cikkszam` BIGINT NOT NULL UNIQUE,
    `ar` INT UNSIGNED NOT NULL
);

CREATE TABLE `szerelo` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    `nev` VARCHAR(255) NOT NULL,
    `munkadij` INT UNSIGNED NOT NULL,
    `szakterulet` VARCHAR(255) NOT NULL
);

CREATE TABLE `javitas` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    `nev` VARCHAR(255) NOT NULL,
    `datum` DATE NOT NULL,
    `munkaIdo` INT UNSIGNED NOT NULL,
    `szereloId` BIGINT UNSIGNED NOT NULL,
    `autoId` BIGINT UNSIGNED NOT NULL
);

CREATE TABLE `javitas_tetel` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    `javitasId` BIGINT UNSIGNED NOT NULL,
    `alkatreszId` BIGINT UNSIGNED NOT NULL,
    `mennyiseg` INT UNSIGNED NOT NULL DEFAULT 1,
    `egysegAr` INT UNSIGNED NOT NULL
);

ALTER TABLE `auto`
    ADD CONSTRAINT `auto_ugyfelid_foreign`
    FOREIGN KEY (`ugyfelId`) REFERENCES `ugyfel`(`id`);

ALTER TABLE `javitas`
    ADD CONSTRAINT `javitas_autoid_foreign`
    FOREIGN KEY (`autoId`) REFERENCES `auto`(`id`);


ALTER TABLE `javitas`
    ADD CONSTRAINT `javitas_szereloid_foreign`
    FOREIGN KEY (`szereloId`) REFERENCES `szerelo`(`id`);

ALTER TABLE `javitas_tetel`
    ADD CONSTRAINT `javitas_tetel_javitasid_foreign`
    FOREIGN KEY (`javitasId`) REFERENCES `javitas`(`id`);

ALTER TABLE `javitas_tetel`
    ADD CONSTRAINT `javitas_tetel_alkatreszid_foreign`
    FOREIGN KEY (`alkatreszId`) REFERENCES `alkatresz`(`id`);

SELECT ugyfel.nev, COUNT(auto.id) AS autok_szama
FROM ugyfel
JOIN auto ON auto.ugyfelId = ugyfel.id
GROUP BY ugyfel.id, ugyfel.nev
ORDER BY autok_szama DESC
LIMIT 1;

SELECT auto.rendszam, COUNT(javitas.id) AS szervizek_szama
FROM auto
JOIN javitas ON javitas.autoId = auto.id
GROUP BY auto.id, auto.rendszam
ORDER BY szervizek_szama DESC
LIMIT 1;

SELECT szerelo.nev, SUM(javitas.munkaIdo) AS osszes_munkaido
FROM szerelo
JOIN javitas ON javitas.szereloId = szerelo.id
GROUP BY szerelo.id, szerelo.nev
ORDER BY osszes_munkaido DESC
LIMIT 1;

SELECT 
    javitas.id,
    javitas.nev,
    (javitas.munkaIdo * szerelo.munkadij + COALESCE(SUM(javitas_tetel.mennyiseg * javitas_tetel.egysegAr), 0)) AS teljes_koltseg
FROM javitas
JOIN szerelo ON javitas.szereloId = szerelo.id
LEFT JOIN javitas_tetel ON javitas_tetel.javitasId = javitas.id
GROUP BY javitas.id, javitas.nev, javitas.munkaIdo, szerelo.munkadij
ORDER BY teljes_koltseg DESC
LIMIT 1;

SELECT alkatresz.nev, COUNT(javitas_tetel.id) AS felhasznalasok_szama
FROM alkatresz
JOIN javitas_tetel ON javitas_tetel.alkatreszId = alkatresz.id
GROUP BY alkatresz.id, alkatresz.nev
ORDER BY felhasznalasok_szama DESC
LIMIT 1;

SELECT auto.rendszam
FROM auto
LEFT JOIN javitas ON javitas.autoId = auto.id
WHERE javitas.id IS NULL;

CREATE VIEW szereloi_teljesitmeny AS
SELECT
    szerelo.id AS szerelo_id,
    szerelo.nev AS szerelo_nev,
    szerelo.szakterulet,
    COUNT(javitas.id) AS javitasok_szama,
    COALESCE(SUM(javitas.munkaIdo), 0) AS osszes_munkaora,
    COALESCE(SUM(javitas.munkaIdo * szerelo.munkadij), 0) AS munkadij_bevetel
FROM szerelo
LEFT JOIN javitas ON javitas.szereloId = szerelo.id
GROUP BY szerelo.id, szerelo.nev, szerelo.szakterulet, szerelo.munkadij;

CREATE VIEW autok_osszkoltsege AS
SELECT
    auto.id AS auto_id,
    auto.rendszam,
    ugyfel.nev AS tulajdonos,
    COUNT(DISTINCT javitas.id) AS javitasok_szama,
    COALESCE(SUM(DISTINCT javitas.id * 0 + javitas.munkaIdo * szerelo.munkadij), 0) AS osszes_munkadij,
    COALESCE(SUM(javitas_tetel.mennyiseg * javitas_tetel.egysegAr), 0) AS osszes_alkatreszkoltseg,
    COALESCE(SUM(DISTINCT javitas.id * 0 + javitas.munkaIdo * szerelo.munkadij), 0)
    + COALESCE(SUM(javitas_tetel.mennyiseg * javitas_tetel.egysegAr), 0) AS teljes_osszeg
FROM auto
JOIN ugyfel ON auto.ugyfelId = ugyfel.id
LEFT JOIN javitas ON javitas.autoId = auto.id
LEFT JOIN szerelo ON javitas.szereloId = szerelo.id
LEFT JOIN javitas_tetel ON javitas_tetel.javitasId = javitas.id
GROUP BY auto.id, auto.rendszam, ugyfel.nev;

CREATE VIEW munkalapok_reszletes AS
SELECT
    javitas.id AS munkalap_id,
    javitas.nev AS munkalap_nev,
    javitas.datum,
    javitas.munkaIdo,
    auto.rendszam,
    ugyfel.nev AS tulajdonos,
    szerelo.nev AS szerelo,
    alkatresz.nev AS alkatresz,
    javitas_tetel.mennyiseg,
    javitas_tetel.egysegAr,
    (javitas_tetel.mennyiseg * javitas_tetel.egysegAr) AS tetel_osszeg
FROM javitas
JOIN auto ON javitas.autoId = auto.id
JOIN ugyfel ON auto.ugyfelId = ugyfel.id
JOIN szerelo ON javitas.szereloId = szerelo.id
LEFT JOIN javitas_tetel ON javitas_tetel.javitasId = javitas.id
LEFT JOIN alkatresz ON javitas_tetel.alkatreszId = alkatresz.id;
